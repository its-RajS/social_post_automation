How to Start the System

cd automation/

# 1. Start infra (postgres, redis, minio, chroma)
docker compose up -d postgres redis minio chroma

# 2. Run DB migration (first time only)
cd nodejs-api && DATABASE_URL="postgresql://user:pass@localhost:5432/linkedin_automation" npx prisma migrate deploy && cd ..

# 3. Start everything
docker compose up --build -d

# 4. Verify
curl localhost:3000/health          # → {"ok":true}
curl localhost:8001/health          # → {"ok":true,"redis":true,"db":true}
